# HTTP proxy curl -x http://USER:PASS@gate.packetshift.io:8080 https://api.ipify.org # SOCKS5 curl --socks5 USER:PASS@gate.packetshift.io:1080 https://api.ipify.org # Country + sticky session curl -x http://USER-country-us-session-abc1:PASS@gate.packetshift.io:8080 https://api.ipify.org
import requests proxies = { "http": "http://USER:PASS@gate.packetshift.io:8080", "https": "http://USER:PASS@gate.packetshift.io:8080", } # Rotating — new IP every request r = requests.get("https://api.ipify.org", proxies=proxies) print(r.text) # → 185.220.101.47 # Sticky — same IP for 10 min sticky = "http://USER-session-xyz123:PASS@gate.packetshift.io:8080"
const axios = require('axios') const { HttpsProxyAgent } = require('https-proxy-agent') const agent = new HttpsProxyAgent( `http://USER:PASS@gate.packetshift.io:8080` ) const res = await axios.get('https://api.ipify.org', { httpsAgent: agent, httpAgent: agent }) console.log(res.data) // → 185.220.101.47
